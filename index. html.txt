    <script>
        let chosenGame = 'Free Fire';
        let chosenPkg = '100 ពេជ្រ/UC';

        function selectGame(game, event) {
            chosenGame = game;
            document.getElementById('selectedGame').value = game;
            let cards = document.querySelectorAll('.game-card');
            cards.forEach(c => c.classList.remove('active'));
            event.currentTarget.classList.add('active');
        }

        function selectPkg(element, pkg) {
            chosenPkg = pkg;
            let btns = document.querySelectorAll('.pkg-btn');
            btns.forEach(b => b.classList.remove('selected'));
            element.classList.add('selected');
        }

        function checkout() {
            let id = document.getElementById('playerId').value;
            if(!id) {
                alert('សូមបញ្ចូល Player ID របស់អ្នកសិន!');
                return;
            }
            alert(`បញ្ជាទិញជោគជ័យ!\nវេបសាយ: Y2K Top Up\nហ្គេម: ${chosenGame}\nID: ${id}\nកញ្ចប់: ${chosenPkg}\nសូមទាក់ទងមក Telegram ដើម្បីបង់ប្រាក់។`);
        }
